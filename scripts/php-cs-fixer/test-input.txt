<?php

for ($i = 0; $i < 10; $i++) {
  echo $i . "\n";
}

for ($j = 0; $j < 5; $j++) {
  echo $j . "\n";
}

if (isset($tokens)) {
  unset($tokens);
}
$tokens = [];
gc_collect_cycles();

for ($index = $tokens->count() - 1; $index >= 0; $index--) {
  $token = $tokens[$index];
  echo $token->getContent() . "\n";
}
